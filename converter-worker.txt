// Local-only FFmpeg worker. It receives the selected video as bytes; it never
// performs a network request or receives a remote URL to the user's media.
self.onmessage = async ({data}) => {
  try {
    if (!self.createFFmpegCore) throw new Error('Конвертер не загрузился.');
    const ff = await self.createFFmpegCore({wasmBinary:new Uint8Array(data.wasm)});
    ff.setLogger(({message}) => {
      if (/invalid data|unknown decoder|error while|error opening|no such file|not found/i.test(message)) {
        self.postMessage({type:'diagnostic',text:String(message).slice(0,240)});
      }
    });
    ff.setProgress(({progress}) => {
      if (Number.isFinite(progress)) self.postMessage({type:'progress',value:Math.max(0,Math.min(1,progress))});
    });
    const match=String(data.name||'').toLowerCase().match(/\.([a-z0-9]{1,8})$/);
    const input='source.'+(match ? match[1] : 'media');
    ff.FS.writeFile(input,new Uint8Array(data.bytes));
    // Preserve the complete video; map at most the first video and audio tracks.
    // Re-encode because changing the container alone cannot repair incompatible codecs.
    ff.exec('-i',input,'-map','0:v:0','-map','0:a:0?',
      '-vf','pad=ceil(iw/2)*2:ceil(ih/2)*2',
      '-c:v','libx264','-preset','ultrafast','-crf','20','-pix_fmt','yuv420p',
      '-c:a','aac','-b:a','128k','-movflags','+faststart','-threads','1','prepared.mp4');
    if (ff.ret !== 0) throw new Error('Видео не удалось преобразовать (код '+ff.ret+').');
    const out=ff.FS.readFile('prepared.mp4');
    if (!out || !out.length) throw new Error('Конвертер вернул пустой файл.');
    self.postMessage({type:'done',bytes:out},[out.buffer]);
  } catch (error) {
    self.postMessage({type:'error',text:String(error?.message || error)});
  }
};
